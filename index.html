function radiateLettersWithBeams(){ const centerX = window.innerWidth / 2; const centerY = window.innerHeight / 2; const lettersArr = ['P','R','I','N','C','E']; lettersArr.forEach((l, i) => { const el = document.createElement('div'); el.className = 'letter'; el.innerText = l; el.style.color = '#c0c0c0'; // steel metallic color el.style.textShadow = '0 0 12px #fff,0 0 20px #aaa'; stage.appendChild(el); el.style.left = centerX + 'px'; el.style.top = centerY + 'px'; el.style.opacity = 0; el.style.transform = 'translate(-50%,-50%) scale(0)'; const angle = (i - (lettersArr.length-1)/2) * 20; // spread const radius = 300; // Beam animation setTimeout(() => { el.style.transition = 'transform 1.6s cubic-bezier(0.22,1,0.36,1), opacity 1.4s ease-out'; el.style.opacity = 1; el.style.transform = ` translate(${Math.cos(angle*Math.PI/180)*radius}px, ${Math.sin(angle*Math.PI/180)*radius}px) scale(1.2) rotate(${(Math.random()-0.5)*20}deg) `; // Optional: add a glowing "beam" behind the letter const beam = document.createElement('div'); beam.style.position = 'absolute'; beam.style.left = centerX + 'px'; beam.style.top = centerY + 'px'; beam.style.width = '2px'; beam.style.height = radius + 'px'; beam.style.background = 'linear-gradient(to bottom, #fff8, transparent)'; beam.style.transform = `rotate(${angle}deg) translate(-50%,-50%)`; beam.style.zIndex = 20; stage.appendChild(beam); setTimeout(() => { beam.remove(); // beam disappears after animation }, 1600); }, i*200); // Settle at bottom of star setTimeout(() => { el.style.transition = 'transform 1.5s cubic-bezier(0.68,-0.55,0.27,1.55), opacity 1s ease-out'; el.style.transform = `translate(${(i-(lettersArr.length-1)/2)*12}vw, 80vh) scale(1) rotate(0deg)`; }, 2200 + i*200); }); }
